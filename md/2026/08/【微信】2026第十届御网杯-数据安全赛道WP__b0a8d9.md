```python
import csv
import hashlib
from collections import defaultdict, Counter

base = r'D:\下载\8：TracePurge的附件\tempdir\DS附件\TracePurge附件\TracePurge附件'
out_path = r'D:\下载\8：TracePurge的附件\tempdir\DS附件\TracePurge附件\TracePurge附件\response_plan.csv'

leak_path = base + r'\leak_sample.csv'
idx_path = base + r'\customer_index.csv'

target_batch = 'CRM-20260508-D4'
export_id = 'EX20260508004'
operator = 'ops_chen'

email_map = {}
name_phone = defaultdict(list)

with open(idx_path, newline='', encoding='utf-8-sig') as f:
    for row in csv.DictReader(f):
        if row['email_hash']:
            email_map[row['email_hash']] = row
        name_phone[(row['name'], row['phone_mask'])].append(row)

def match_subject(leak):
    email = leak['email'].strip()
    if email:
        h = hashlib.sha256(email.lower().encode()).hexdigest()
        if h in email_map:
            return email_map[h]

    hits = name_phone[(leak['name'], leak['phone_mask'])]
    if len(hits) == 1:
        return hits[0]

    returnNone

def get_action(subject):
    if subject['erase_requested'] == 'true':
        return'PURGE'
    if subject['risk'] == 'high':
        return'NOTIFY'
    return'MONITOR'

best = {}

with open(leak_path, newline='', encoding='utf-8-sig') as f:
    for leak in csv.DictReader(f):
        if leak['batch_tag'] != target_batch:
            continue

        subject = match_subject(leak)
        ifnot subject:
            continue

        sid = subject['subject_id']

        row = {
            'leak_id': leak['leak_id'],
            'subject_id': sid,
            'export_id': export_id,
            'operator': operator,
            'action': get_action(subject),
        }

        if sid notin best or leak['leak_id'] < best[sid]['leak_id']:
            best[sid] = row

rows = sorted(best.values(), key=lambda r: (r['leak_id'], r['subject_id']))

with open(out_path, 'w', newline='', encoding='utf-8') as f:
    writer = csv.DictWriter(
        f,
        fieldnames=['leak_id', 'subject_id', 'export_id', 'operator', 'action']
    )
    writer.writeheader()
    writer.writerows(rows)

print('output:', out_path)
print('rows:', len(rows))
print(Counter(r['action'] for r in rows))
```

```python
err = pathlib.Path(r"D:\下载\3：ShadowMeter的附件\tempdir\MISC附件\ShadowMeter附件\ShadowMeter附件\error.log")
out = pathlib.Path(r"D:\下载\3：ShadowMeter的附件\tempdir\MISC附件\ShadowMeter附件\ShadowMeter附件\1.zip")

line_re = re.compile(
    r"^\[(?P<ts>[^\]]+)\].*?\[client (?P<client>[^\]]+)\] "
    r"mod_dumpio:\s+dumpio_(?P<dir>in|out) \((?P<kind>[^)]+)\): (?P<data>.*)$"
)

def ts(s):
    return datetime.datetime.strptime(s, "%a %b %d %H:%M:%S.%f %Y")

def unesc(s):
    return codecs.decode(s.encode(), "unicode_escape").encode("latin-1", "replace")

streams = collections.defaultdict(list)

for line in err.read_text(errors="replace").splitlines():
    m = line_re.match(line)
    ifnot m:
        continue
    if m.group("dir") != "in":
        continue
    ifnot m.group("kind").startswith("data-"):
        continue
    data = m.group("data")
    if re.fullmatch(r"\d+ bytes", data):
        continue
    streams[m.group("client")].append((ts(m.group("ts")), unesc(data)))

def parse_requests(buf):
    pos = 0
    while pos < len(buf):
        end = buf.find(b"\r\n\r\n", pos)
        if end < 0:
            break
        head = buf[pos:end]
        first = head.split(b"\r\n", 1)[0].decode("latin-1", "replace")
        m = re.match(r"(GET|POST|OPTIONS) ([^ ]+) HTTP/", first)
        ifnot m:
            nxt = buf.find(b"POST ", pos + 1)
            if nxt < 0:
                break
            pos = nxt
            continue
        headers = {}
        for line in head.split(b"\r\n")[1:]:
            ifb":"in line:
                k, v = line.split(b":", 1)
                headers[k.decode().lower()] = v.strip().decode()
        clen = int(headers.get("content-length", "0") or0)
        body = buf[end + 4:end + 4 + clen]
        yield m.group(1), m.group(2), body
        pos = end + 4 + clen

rows = []

for parts in streams.values():
    buf = b"".join(x[1] for x in sorted(parts))
    for method, path, body in parse_requests(buf):
        if path.split("?")[0] != "/collect/report.php":
            continue
        qs = urllib.parse.parse_qs(body.decode("latin-1"), keep_blank_values=True)
        rows.append({k: v[0] for k, v in qs.items()})

sm = [r for r in rows if r.get("job") == "SM-20260507-17"]
end_note = next(r["note"] for r in sm if r.get("seq") == "END")
target_size = int(re.search(r"archive_size=(\d+)", end_note).group(1))
target_sha = re.search(r"sha256=([0-9a-f]+)", end_note).group(1)

chunks = {}
for r in sm:
    seq = r.get("seq", "")
    if seq.isdigit():
        chunks.setdefault(int(seq), r["note"])

b64 = "".join(chunks[i] for i in range(1, max(chunks) + 1))
data = base64.b64decode(b64)

print(len(data))
print(hashlib.sha256(data).hexdigest())

assert len(data) == target_size
assert hashlib.sha256(data).hexdigest() == target_sha

out.write_bytes(data)
print(out)
```

## peach_garden_xor

![图片](https://cdn.jsdelivr.net/gh/zhiyu-zeng/img@main/img/2026/08/91bb3c5f5f250497.png)


然后想一下，.docx 本质是 zip，zip 文件头通常是：

```
50 4b 03 04 14 00 00 00
```

所以密钥可以用密文 XOR 明文头推出：

```python
c = bytes.fromhex("51 7e 7f ad 15 35 7c a9") p = bytes.fromhex("50 4b 03 04 14 00 00 00") print(bytes(a ^ b for a, b in zip(c, p)).hex())
```

```python
from pathlib import Path
import zipfile

enc = Path(r'.\task.docx.enc')
out = Path(r'.\task.docx')
key = bytes.fromhex('01357ca9')

data = enc.read_bytes()
plain = bytes(b ^ key[i % len(key)] for i, b in enumerate(data))
out.write_bytes(plain)

print(plain[:16].hex())
print(zipfile.is_zipfile(out))
```

```python
from pathlib import Path
from PIL import Image

def arnold_inv(img, n=7):
    w = img.width
    cur = img.copy()
    for _ in range(n):
        old = cur.copy()
        ps = old.load()
        pd = cur.load()
        for x in range(w):
            for y in range(w):
                pd[(2 * x - y) % w, (-x + y) % w] = ps[x, y]
    return cur

def extract_lsb(img):
    px = img.load()
    w, h = img.size
    bits = []
    for y in range(h):
        for x in range(w):
            r, g, b = px[x, y][:3]
            bits += [r & 1, g & 1, b & 1]
            if len(bits) >= 32:
                size = 0
                for bit in bits[:32]:
                    size = (size << 1) | bit
                need = (size + 4) * 8
                if len(bits) >= need:
                    raw = bytearray()
                    for i in range(0, need, 8):
                        v = 0
                        for bit in bits[i:i + 8]:
                            v = (v << 1) | bit
                        raw.append(v)
                    return size, bytes(raw[4:])

img = Image.open("poster.png").convert("RGB")

Path("1.png").write_bytes(payload)

print("saved: 1.png")
```

```python
def valid_idcard(value: str) -> bool:
    ifnot re.fullmatch(r"\d{17}[\dX]", value):
        returnFalse
    birth = value[6:14]
    try:
        dt = datetime.strptime(birth, "%Y%m%d")
    except ValueError:
        returnFalse
    if dt.year < 1900or dt.year > 2030:
        returnFalse
    checksum = sum(int(ch) * weight for ch, weight in zip(value[:17], ID_WEIGHTS))
    return ID_CHECK[checksum % 11] == value[-1]


def valid_luhn(value: str) -> bool:
    ifnot re.fullmatch(r"\d{16,19}", value):
        returnFalse
    total = 0
    for idx, ch in enumerate(reversed(value), 1):
        digit = int(ch)
        if idx % 2 == 0:
            digit *= 2
            if digit > 9:
                digit -= 9
        total += digit
    return total % 10 == 0


def valid_phone(value: str) -> bool:
    return bool(re.fullmatch(r"\d{11}", value)) and value[:3] in PHONE_PREFIXES


def valid_ip(value: str) -> bool:
    parts = value.split(".")
    if len(parts) != 4:
        returnFalse
    for part in parts:
        ifnot re.fullmatch(r"\d+", part):
            returnFalse
        if len(part) > 1and part.startswith("0"):
            returnFalse
        if int(part) > 255:
            returnFalse
    returnTrue


def classify(value: str) -> tuple[str, str, str]:
    v = id_candidate(value)
    if valid_idcard(v):
        return v, "idcard", "S4"

    v = bank_candidate(value)
    if valid_luhn(v):
        return v, "bankcard", "S4"

    v = phone_candidate(value)
    if valid_phone(v):
        return v, "phone", "S3"

    v = ip_candidate(value)
    if valid_ip(v):
        return v, "ip", "S2"

    return normal_candidate(value), "normal", "S1"

    ) as dst:
        reader = csv.DictReader(src)
        writer = csv.DictWriter(dst, fieldnames=["data_id", "value", "category", "level"])
        writer.writeheader()
        for row in reader:
            plain = decrypt_if_needed(row["value"], key)
            normalized, category, level = classify(plain)
            writer.writerow(
                {
                    "data_id": row["data_id"],
                    "value": normalized,
                    "category": category,
                    "level": level,
                }
            )


if __name__ == "__main__":
    main()
```

```sql
WITH ghost_accounts AS (
    SELECT
        u.id AS user_id,
        u.username,
        u.name,
        u.idcard,
        u.phone,
        u.role_id,
        r.role_name,
        u.status,
        CASE
            WHEN is_valid_idcard(u.idcard) = 0AND is_valid_phone(u.phone) = 0THEN'idcard,phone'
            WHEN is_valid_idcard(u.idcard) = 0THEN'idcard'
            ELSE'phone'
        ENDAS invalid_fields
    FROMusersAS u
    JOINrolesAS r
        ON r.id = u.role_id
    WHERE is_valid_idcard(u.idcard) = 0
       OR is_valid_phone(u.phone) = 0
),
authorized_permissions AS (
    SELECT
        role_id,
        api_endpoint
    FROM api_permissions
    WHERE can_access = 1
),
violation_candidates AS (
    SELECT
        l.id AS log_id,
        l.user_id,
        u.username,
        u.name,
        u.status,
        u.role_id,
        r.role_name,

            WHEN u.status = 'inactive'THEN1
            WHENtime(l.access_time) < '09:00:00'ORtime(l.access_time) > '18:00:00'THEN2
            WHEN ap.role_id ISNULLTHEN3

            CASEWHEN u.status = 'inactive'THEN'1,'ELSE''END ||
            CASEWHENtime(l.access_time) < '09:00:00'ORtime(l.access_time) > '18:00:00'THEN'2,'ELSE''END ||
            CASEWHEN ap.role_id ISNULLTHEN'3,'ELSE''END,

        ON u.id = l.user_id

        ON r.id = u.role_id
    LEFTJOIN authorized_permissions AS ap
        ON ap.role_id = u.role_id
       AND ap.api_endpoint = l.api_endpoint
    WHERE u.id NOTIN (SELECT user_id FROM ghost_accounts)

            u.status = 'inactive'
```

### CPA 攻击

AES 首轮中，每个字节会经过：

```
SBox(P[i] ^ K[i])
```

其中 `P[i]` 是已知明文字节， `K[i]` 是要猜的密钥字节。CPA 的做法是：

1.  对某个 key byte 枚举 `0x00` 到 `0xff` 。
    
2.  对每条明文计算假设中间值。
    
3.  把假设泄漏值和真实迹线逐点做 Pearson 相关。
    
4.  哪个 key guess 的相关性最高，就认为哪个是正确密钥字节。
    

尝试最常见模型：

```
HW(SBox(P ^ K))
```

但这个模型的平均相关系数只有约 0.156，说明能看到一点信号，但不是设备真正主要泄漏的量

![图片](https://cdn.jsdelivr.net/gh/zhiyu-zeng/img@main/img/2026/08/43d9777d519483e4.png)

真正有效的模型是

```
HW(SBox(P ^ K) ^ P)
```

这个模型表示 SubBytes 前后的 Hamming Distance，也就是状态从 `P` 变成 `SBox(P ^ K)` 时翻转了多少 bit。对功耗/电磁侧信道来说，总线或寄存器翻转数量经常比单纯的 Hamming Weight 更明显。

![图片](https://cdn.jsdelivr.net/gh/zhiyu-zeng/img@main/img/2026/08/2f8d039a555bcdac.png)

恢复出的 AES master key 为：

```
422c174a4a89d96af0b7de281dc01324
```

直接用master key去尝试flag.enc并不能得到正确明文。继续做 AES-128 key schedule，可以得到第 10 轮密钥

flag.enc实际使用round10_key作为AES-ECB key加密。用该key解密并去掉PKCS#7 padding

```python
    traces = np.load(data_dir / "traces.npy")
    flag_enc = (data_dir / "flag.enc").read_bytes()
    return plaintexts, traces, flag_enc


def centered_trace_channel(traces: np.ndarray, channel: int = 0) -> tuple[np.ndarray, np.ndarray]:
    matrix = traces[:, channel, :].astype(np.float32)
    matrix -= matrix.mean(axis=0, keepdims=True)
    norm = np.sqrt((matrix * matrix).sum(axis=0))
    norm[norm == 0] = 1
    return matrix, norm


def build_leakage(plain_byte: np.ndarray, guesses: np.ndarray, model: str) -> np.ndarray:
    state = np.bitwise_xor(plain_byte[:, None], guesses)
    sbox_value = SBOX[state]

    if model == "HW(SBox(P^K))":
        leakage = HW[sbox_value]
    elif model == "HW(SBox(P^K)^P)":
        leakage = HW[np.bitwise_xor(sbox_value, plain_byte[:, None])]
    else:
        raise ValueError(model)

    leakage = leakage.T.astype(np.float32)
    leakage -= leakage.mean(axis=1, keepdims=True)
    return leakage


def run_cpa(plaintexts: np.ndarray, traces: np.ndarray, model: str) -> tuple[bytes, list[dict]]:
    trace_matrix, trace_norm = centered_trace_channel(traces, channel=0)
    guesses = np.arange(256, dtype=np.uint8)
    key = []
    table = []

    for byte_index in range(16):
        leakage = build_leakage(plaintexts[:, byte_index], guesses, model)
        leakage_norm = np.sqrt((leakage * leakage).sum(axis=1))
        leakage_norm[leakage_norm == 0] = 1

        corr = (leakage @ trace_matrix) / (leakage_norm[:, None] * trace_norm[None, :])
        abs_corr = np.abs(corr)
        score = abs_corr.max(axis=1)
        order = np.argsort(score)[::-1]

        best = int(order[0])
        second = int(order[1])
        sample = int(abs_corr[best].argmax())
        key.append(best)
        table.append(
            {
                "byte": byte_index,
                "best": best,
                "corr": float(score[best]),
                "sample": sample,
                "second": second,
                "second_corr": float(score[second]),
            }
        )

    return bytes(key), table


def expand_aes128(master_key: bytes) -> list[bytes]:
    def xor_word(a: bytes, b: bytes) -> bytes:
        return bytes(x ^ y for x, y in zip(a, b))

    def sub_word(word: bytes) -> bytes:
        return bytes(int(SBOX[b]) for b in word)

    def rot_word(word: bytes) -> bytes:
        return word[1:] + word[:1]

    words = [master_key[i : i + 4] for i in range(0, 16, 4)]
    for i in range(4, 44):
        temp = words[i - 1]
        if i % 4 == 0:
            temp = sub_word(rot_word(temp))
            temp = bytes([temp[0] ^ RCON[i // 4]]) + temp[1:]
        words.append(xor_word(words[i - 4], temp))

    return [b"".join(words[4 * r : 4 * r + 4]) for r in range(11)]
```

## close primes

先解析公钥，得到 1024-bit RSA，公钥指数为常见的 `65537` 。密文文件大小为 `23168` 字节，而 1024-bit RSA 的单个密文块长度是 `128` 字节

```
23168 / 128 = 181
```

所以 `encrypted.bin` 不是一个单独 RSA 密文，而是由 181 个 RSA 密文块拼接而成。后面恢复私钥后，需要按 128 字节分块逐块解密。

Fermat 分解

设：

```toml
n = p * q
a = (p + q) / 2
b = (q - p) / 2
```

则：

```
n = a^2 - b^2 = (a - b)(a + b)
```

也就是说，从 `ceil(sqrt(n))` 开始枚举 `a` ，只要发现：

```
a^2 - n = b^2
```

就可以得到：

```
p = a - b
q = a + b
```

`p` 和 `q` 的距离确实很小，Fermat 分解一次就能命中

```
p = 12751828551709836069103027639236748210345732901386209604520393543958229433834765694925409378355486164691077551258041029878391501131185664614971880587757401
q = 12751828551709836069103027639236748210345732901386209604520393543958229433834765694925409378355486164691077551258041029878391501131185675868445062344890849
q - p = 11253473181757133448
```

恢复 `p` 、 `q` 后，就可以计算：

```
phi(n) = (p - 1) * (q - 1)
d = e^{-1} mod phi(n)
```

然后构造 RSA 私钥。

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
from math import isqrt
from pathlib import Path

from Crypto.Cipher import PKCS1_OAEP
from Crypto.Hash import SHA1, SHA224, SHA256, SHA384, SHA512
from Crypto.PublicKey import RSA
from Crypto.Util.number import inverse


HASH_CANDIDATES = [
    ("SHA1", SHA1),
    ("SHA224", SHA224),
    ("SHA256", SHA256),
    ("SHA384", SHA384),
    ("SHA512", SHA512),
]


def fermat_factor(n: int) -> tuple[int, int]:
    a = isqrt(n)
    if a * a < n:
        a += 1

    whileTrue:
        b2 = a * a - n
        b = isqrt(b2)
        if b * b == b2:
            p = a - b
            q = a + b
            if p * q == n:
                return int(p), int(q)
        a += 1


def split_blocks(data: bytes, block_size: int) -> list[bytes]:
    if len(data) % block_size != 0:
        raise ValueError(
            f"ciphertext length {len(data)} is not a multiple of block size {block_size}"
        )
    return [data[i : i + block_size] for i in range(0, len(data), block_size)]


def decrypt_oaep_blocks(
    key: RSA.RsaKey, blocks: list[bytes]
) -> tuple[str, bytes] | None:
    for hash_name, hash_mod in HASH_CANDIDATES:
        cipher = PKCS1_OAEP.new(key, hashAlgo=hash_mod)
        plaintext_parts = []
        try:
            for block in blocks:
                plaintext_parts.append(cipher.decrypt(block))
        except ValueError:
            continue
        return hash_name, b"".join(plaintext_parts)
    returnNone


def guess_extension(blob: bytes) -> str:
    if blob.startswith(b"\x89PNG\r\n\x1a\n"):
        return".png"
    if blob.startswith(b"\xff\xd8\xff"):
        return".jpg"
    if blob.startswith(b"GIF87a") or blob.startswith(b"GIF89a"):
        return".gif"
    if blob.startswith(b"BM"):
        return".bmp"
    if blob.startswith(b"RIFF") and blob[8:12] == b"WEBP":
        return".webp"
    return".bin"


def solve(input_dir: Path, output_path: Path | None) -> tuple[Path, dict[str, int | str]]:
    input_dir = input_dir.resolve()
    pubkey_path = input_dir / "pubkey.pem"
    encrypted_path = input_dir / "encrypted.bin"

    public_key = RSA.import_key(pubkey_path.read_bytes())
    n = public_key.n
    e = public_key.e

    p, q = fermat_factor(n)
    phi = (p - 1) * (q - 1)
    d = inverse(e, phi)
    private_key = RSA.construct((n, e, d, p, q))

    ciphertext = encrypted_path.read_bytes()
    blocks = split_blocks(ciphertext, public_key.size_in_bytes())

    decrypted = decrypt_oaep_blocks(private_key, blocks)
    if decrypted isNone:
        raise RuntimeError("unable to decrypt all blocks with common OAEP hash choices")

    hash_name, plaintext = decrypted
    if output_path isNone:
        output_path = input_dir / f"decrypted_image{guess_extension(plaintext)}"

    output_path.write_bytes(plaintext)

    metadata: dict[str, int | str] = {
        "n_bits": public_key.size_in_bits(),
        "e": e,
        "p": p,
        "q": q,
        "prime_gap": q - p,
        "block_size": public_key.size_in_bytes(),
        "block_count": len(blocks),
        "oaep_hash": hash_name,
        "plaintext_len": len(plaintext),
    }
    return output_path, metadata


def main() -> None:
    parser = argparse.ArgumentParser(description="Solve q4_close_primes")
    parser.add_argument(
        "--input-dir",
        type=Path,
        default=Path("."),
        help="directory containing pubkey.pem and encrypted.bin; defaults to current directory",
    )
    parser.add_argument(
        "--output",
        type=Path,
        default=None,
        help="output image path; defaults to decrypted_image.<ext> in the input directory",
    )
    args = parser.parse_args()

    output_path, metadata = solve(args.input_dir, args.output)
    print(f"[+] saved plaintext to: {output_path}")
    for key, value in metadata.items():
        print(f"[+] {key}: {value}")


if __name__ == "__main__":
    main()
```
