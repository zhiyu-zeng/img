这里说一下，不管什么类型文件，它都是有固定格式文件头的，视频也不例外，比如

FLV：前3字节固定为 46 4C 56，对应ASCII字符"FLV"

MPEG系列：标准起始标识为 00 00 01 BA 或 00 00 01 B3

AVI/WAV等RIFF容器：以 52 49 46 46 作为开头标识

function scanStartcodes(file) {
  const fd = fs.openSync(file, 'r');
  const size = fs.fstatSync(fd).size;
  const offs = [];
  let off = 0, tail = Buffer.alloc(0);
  while (off < size) {
    const want = Math.min(BLOCK, size - off);
    const buf = Buffer.allocUnsafe(tail.length + want);
    tail.copy(buf, 0);
    const got = fs.readSync(fd, buf, tail.length, want, off);
    const total = tail.length + got;
    let p = 0;
    while ((p = buf.indexOf(SC, p)) !== -1) {
      if (p + 4 > total) break;
      const abs = off - tail.length + p;
      if (abs >= 0 && abs + 4 <= size) offs.push(abs);
      p += 1;
    }
    const keep = Math.min(3, total);
    tail = Buffer.from(buf.slice(total - keep, total));
    off += got;
    if (!got) break;
  }
  fs.closeSync(fd);
  return { offs, size };
}

function classify(file, offs, size) {
  const fd = fs.openSync(file, 'r');
  const one = Buffer.allocUnsafe(2);
  const stats = new Map();
  const seg = [];
  let priv = 0, privBytes = 0;
  for (let i = 0; i < offs.length; i++) {
    const s = offs[i], e = i + 1 < offs.length ? offs[i + 1] : size;
    const n = fs.readSync(fd, one, 0, 2, s + 4);
    if (n < 1) continue;
    const t = one[0], len = e - s;
    if (t >> 7) { priv++; privBytes += len; }
    const st = stats.get(t) || { count: 0, bytes: 0 };
    st.count++; st.bytes += len;
    stats.set(t, st);
    seg.push({ off: s, tag: t, len });
  }
  fs.closeSync(fd);
  return { seg, stats, priv, privBytes };
}

// ─────────────────────── 音频解码 ───────────────────────
function alaw2pcm(a) {
  a ^= 0x55;
  let t = (a & 0x0f) << 4;
  const seg = (a & 0x70) >> 4;
  if (seg === 0) t += 8;
  else if (seg === 1) t += 0x108;
  else { t += 0x108; t <<= seg - 1; }
  return (a & 0x80) ? t : -t;
}
function mulaw2pcm(u) {
  u = ~u & 0xff;
  const sign = u & 0x80, exp = (u >> 4) & 0x07, mant = u & 0x0f;
  let s = (((mant << 3) + 0x84) << exp) - 0x84;
  return sign ? -s : s;
}
const DECODER = { alaw: alaw2pcm, mulaw: mulaw2pcm };

/**
 * 提取音轨：c7 08 私有头 + 帧号4 + 参数12 + 0x65 标志 + 320 字节 A-law 数据
 * 按帧号排序，帧边界交叉淡化后写出 8kHz mono 16-bit WAV
 */
function extractAudio(file, seg, outWav) {
  const SPD = opt.audioSamples;      // 每帧采样数
  const FADE = opt.audioFade;
  const dec = DECODER[opt.audioCodec];
  const fd = fs.openSync(file, 'r');
  const h = Buffer.allocUnsafe(32);
  const body = Buffer.allocUnsafe(1024);
  const frames = [];
  let fill = 0;

  for (const s of seg) {
    if (s.tag !== 0xc7) continue;
    if (fs.readSync(fd, h, 0, 24, s.off + 4) < 24) continue;
    if (h[1] !== 0x08) continue;                 // 音频子类型
    if (h[18] !== 0x65) { fill++; continue; }    // 段内 +22 的数据标志
    const ctr = h.readUInt32LE(2);
    const start = s.off + 23;
    const avail = Math.min(s.len - 23, 600);
    if (avail < SPD) continue;
    const got = fs.readSync(fd, body, 0, SPD, start);
    if (got < SPD) continue;
    frames.push({ ctr, buf: Buffer.from(body.slice(0, SPD)) });
  }
  fs.closeSync(fd);

  if (!frames.length) return { frames: 0, seconds: 0, wav: null };
  frames.sort((a, b) => a.ctr - b.ctr);

  const total = frames.length * SPD;
  const pcm = Buffer.allocUnsafe(total * 2);
  let w = 0;
  for (let fi = 0; fi < frames.length; fi++) {
    const cur = frames[fi].buf;
    for (let k = 0; k < SPD; k++) {
      let v = dec(cur[k]);
      if (fi > 0 && k < FADE) {
        const g = k / FADE;
        const pv = dec(frames[fi - 1].buf[SPD - FADE + k]);
        v = Math.round(pv * (1 - g) + v * g);
      }
      if (k >= SPD - FADE && fi < frames.length - 1) {
        v = Math.round(v * (0.5 + 0.5 * (SPD - k) / FADE));
      }
      pcm.writeInt16LE(Math.max(-32768, Math.min(32767, v)), w); w += 2;
    }
  }

  const hdr = Buffer.alloc(44);
  hdr.write('RIFF', 0); hdr.writeUInt32LE(36 + w, 4); hdr.write('WAVE', 8);
  hdr.write('fmt ', 12); hdr.writeUInt32LE(16, 16); hdr.writeUInt16LE(1, 20);
  hdr.writeUInt16LE(1, 22); hdr.writeUInt32LE(8000, 24); hdr.writeUInt32LE(16000, 28);
  hdr.writeUInt16LE(2, 32); hdr.writeUInt16LE(16, 34);
  hdr.write('data', 36); hdr.writeUInt32LE(w, 40);
  const ofd = fs.openSync(outWav, 'w');
  fs.writeSync(ofd, hdr); fs.writeSync(ofd, pcm, 0, w); fs.closeSync(ofd);

  return {
    frames: frames.length, skipped: fill,
    seconds: +(w / 2 / 8000).toFixed(2),
    wav: outWav, mb: +(w / 1048576).toFixed(2),
  };
}

const H264_TAGS = new Set([0x67, 0x68, 0x65, 0x41, 0x61, 0x21, 0x01, 0x05]);
const HEVC_TAGS = new Set([0x40, 0x42, 0x44, 0x26, 0x02, 0x00, 0x01]);

function extractVideo(file, seg, tagSet, outPath, prependParams = false) {
  const fd = fs.openSync(file, 'r');
  const ofd = fs.openSync(outPath, 'w');
  const WB = 8 * 1024 * 1024;
  let wbuf = Buffer.allocUnsafe(WB), wlen = 0;
  const flush = () => { if (wlen) { fs.writeSync(ofd, wbuf, 0, wlen); wlen = 0; } };
  const put = (b) => { if (wlen + b.length > WB) flush(); b.copy(wbuf, wlen); wlen += b.length; };

  if (prependParams) {
    const want = tagSet.has(0x40) ? [0x40, 0x42, 0x44] : [0x67, 0x68];
    const got = new Map();
    for (const s of seg) {
      if (want.includes(s.tag) && !got.has(s.tag)) {
        const b = Buffer.allocUnsafe(Math.min(s.len, 4096));
        const n = fs.readSync(fd, b, 0, b.length, s.off);
        got.set(s.tag, b.slice(0, n));
      }
      if (got.size === want.length) break;
    }
    for (const t of want) if (got.has(t)) put(got.get(t));
  }

  let written = 0, kept = 0;
  for (const s of seg) {
    if (!tagSet.has(s.tag)) continue;
    const n = Math.min(s.len, 16 * 1024 * 1024);
    const b = Buffer.allocUnsafe(n);
    const got = fs.readSync(fd, b, 0, n, s.off);
    put(b.slice(0, got));
    written += got; kept++;
  }
  flush();
  fs.closeSync(ofd); fs.closeSync(fd);
  return { written, kept };
}

// ─────────────────────── POC 重排 ───────────────────────
function parsePoc(buf, off) {
  let pos = (off + 6) * 8;
  const bit = (i) => (buf[i >> 3] >> (7 - (i & 7))) & 1;
  const u = (k) => { let v = 0; for (let i = 0; i < k; i++) { v = (v << 1) | bit(pos); pos++; } return v; };
  const ue = () => { let lz = 0; while (u(1) === 0) { lz++; if (lz > 30) return -1; } return lz === 0 ? 0 : (1 << lz) - 1 + u(lz); };
  u(1); ue(); ue();
  return u(13);
}
function reorderPoc(file, seg, outPath) {
  const size = fs.statSync(file).size;
  const data = fs.readFileSync(file);
  const ofd = fs.openSync(outPath, 'w');
  let gop = [], gops = 0, frames = 0;
  const flushGop = () => {
    if (!gop.length) return;
    gop.sort((a, b) => a.poc - b.poc);
    for (const it of gop) fs.writeSync(ofd, data.slice(it.off, it.end));
    frames += gop.length; gops++; gop = [];
  };
  for (let i = 0; i < seg.length; i++) {
    const s = seg[i];
    const end = i + 1 < seg.length ? seg[i + 1].off : size;
    if (s.tag === 0x26) { flushGop(); fs.writeSync(ofd, data.slice(s.off, end)); }
    else if (s.tag === 0x02) gop.push({ poc: parsePoc(data, s.off), off: s.off, end });
  }
  flushGop();
  fs.closeSync(ofd);
  return { gops, frames };
}

function durToSec(s) {
  if (!s) return 0;
  const parts = s.split(':').map(Number);
  return parts.length === 3 ? parts[0] * 3600 + parts[1] * 60 + parts[2] : 0;
}

// 检查封装结果的音轨是否真的有时长（避免 DURATION 00:00:00 的空音轨）
function audioOk(ffmpeg, file) {
  const r = spawnSync(ffmpeg, ['-hide_banner', '-i', file], { encoding: 'utf8' });
  const txt = (r.stderr || '') + (r.stdout || '');
  if (!/Stream #\d+:\d+.*Audio:/.test(txt)) return false;
  const m = txt.match(/Audio:[^\n]*\n\s*DURATION\s*:\s*([0-9:.]+)/);
  if (m && /^00:00:00(\.0+)?$/.test(m[1])) return false;
  return true;
}

// videoSec: 视频实际时长；用 -t 显式限制，不用 -shortest（后者遇到时间戳异常会把音频截成 0）
function mux(ffmpeg, video, audio, dst, fmt, fps, videoSec) {
  const args = ['-hide_banner', '-loglevel', 'error', '-y', '-fflags', '+genpts', '-r', String(fps), '-i', video];
  if (audio) args.push('-i', audio);
  args.push('-map', '0:v:0');
  if (audio) args.push('-map', '1:a:0');
  args.push('-c:v', 'copy');
  if (audio) {
    // 统一 AAC 48kHz：8kHz PCM 在 MKV 里不少播放器直接静音；
    // 音量 +14dB 因为原始 A-law 平均只有 -33.7dB，再加限幅防爆
    args.push('-af', 'volume=14dB,alimiter=limit=0.95,aresample=48000');
    args.push('-c:a', 'aac', '-b:a', '96k');
    if (videoSec > 0) args.push('-t', videoSec.toFixed(3));
  }
  args.push(dst);
  const r = spawnSync(ffmpeg, args, { encoding: 'utf8', maxBuffer: 64 * 1024 * 1024 });
  return r.status === 0 && fs.existsSync(dst) && fs.statSync(dst).size > 1024;
}

function toMp4Transcode(ffmpeg, video, audio, dst, fps, videoSec) {
  const args = ['-hide_banner', '-loglevel', 'error', '-y', '-r', String(fps), '-i', video];
  if (audio) args.push('-i', audio);
  args.push('-map', '0:v:0');
  if (audio) args.push('-map', '1:a:0');
  args.push('-c:v', 'libx264', '-preset', 'veryfast', '-crf', '20', '-pix_fmt', 'yuv420p');
  if (audio) {
    args.push('-af', 'volume=14dB,alimiter=limit=0.95,aresample=48000');
    args.push('-c:a', 'aac', '-b:a', '96k');
    if (videoSec > 0) args.push('-t', videoSec.toFixed(3));
  }
  args.push('-movflags', '+faststart');
  args.push(dst);
  const r = spawnSync(ffmpeg, args, { encoding: 'utf8', maxBuffer: 64 * 1024 * 1024 });
  return r.status === 0 && fs.existsSync(dst) && fs.statSync(dst).size > 1024;
}

// ─────────────────────── 输出名 / 目录 ───────────────────────
function outName(dir, stem, suffix, ext) {
  let p = path.join(dir, `${stem}${suffix}.${ext}`);
  let i = 2;
  while (fs.existsSync(p)) p = path.join(dir, `${stem}${suffix}_${i++}.${ext}`);
  return p;
}
function ensureOutDir(dir) {
  try {
    fs.mkdirSync(dir, { recursive: true });
    const t = path.join(dir, '.write_test'); fs.writeFileSync(t, 'x'); fs.unlinkSync(t);
    return dir;
  } catch (e) {
    const fb = path.join(process.cwd(), '恢复');
    try {
      fs.mkdirSync(fb, { recursive: true });
      const t = path.join(fb, '.write_test'); fs.writeFileSync(t, 'x'); fs.unlinkSync(t);
      console.error(`[警告] 输出目录 ${dir} 不可用（${e.code || e.message}），已回退到 ${fb}`);
      return fb;
    } catch (e2) {
      console.error(`无法创建输出目录：${dir} → ${e.message}；回退 ${fb} 也失败`);
      process.exit(1);
    }
  }
}

// ─────────────────────── 单文件处理 ───────────────────────
function recoverOne(file, ffmpeg) {
  const t0 = Date.now();
  const stem = path.basename(file, path.extname(file));
  const res = { file, stem, ok: false, branches: [], error: null, audio: null, seconds: 0 };

  const { offs, size } = scanStartcodes(file);
  const { seg, stats, priv, privBytes } = classify(file, offs, size);
  const d = diagnose(stats);
  res.sizeMB = +mb(size);
  res.startcodes = offs.length;
  res.privateHeaders = priv;
  res.detected = d;

  if (opt.probe) {
    res.ok = true;
    res.tags = [...stats.entries()].sort((a, b) => b[1].bytes - a[1].bytes).slice(0, 8)
      .map(([t, s]) => ({ tag: '0x' + t.toString(16).padStart(2, '0'), count: s.count, mb: +mb(s.bytes) }));
    res.seconds = +((Date.now() - t0) / 1000).toFixed(1);
    return res;
  }

  if (!d.h264 && !d.hevc) { res.error = '无可识别的 H.264/HEVC 参数集'; return res; }

  // ── 音轨
  let audioWav = null, audioSec = 0;
  if (opt.audio && d.audio) {
    const w = outName(opt.out, stem, '_audio', 'wav');
    const a = extractAudio(file, seg, w);
    if (a.frames > 0) {
      audioWav = w; audioSec = a.seconds;
      res.audio = { frames: a.frames, seconds: a.seconds, codec: opt.audioCodec, wav: w, mb: a.mb };
    }
  }

  const mode = opt.mode === 'auto' ? (d.h264 ? 'h264' : 'hevc') : opt.mode;

  const buildBranch = (name, tagSet, prepend, outSuffix) => {
    const raw = outName(opt.out, stem, '', name === 'h264' ? 'h264' : 'h265');
    const r = extractVideo(file, seg, tagSet, raw, prepend);

    let src = raw, degraded = false;
    // 只在显式 --mode idr 时降级；自动降级会误伤"部分损坏但大部分可看"的流
    if (name === 'hevc' && opt.mode === 'idr') {
      const idr = outName(opt.out, stem, '_hevc_idr', 'h265');
      extractVideo(file, seg, new Set([0x40, 0x42, 0x44, 0x26]), idr, true);
      src = idr; degraded = true;
    }
    if (opt.reorder && name === 'hevc') {
      const re = outName(opt.out, stem, '_hevc_reordered', 'h265');
      reorderPoc(file, seg, re); src = re;
    }

    const p = probeMedia(ffmpeg, src);
    let vSec = durToSec(p.duration);
    // 裸流 probe 不出 Duration，先封一个临时 MKV 拿真实时长
    if (!vSec) {
      const tmp = outName(opt.out, stem, '_probe_tmp', 'mkv');
      if (mux(ffmpeg, src, null, tmp, 'mkv', opt.fps)) {
        vSec = durToSec(probeMedia(ffmpeg, tmp).duration);
      }
      try { fs.unlinkSync(tmp); } catch (_) {}
    }
    // 音频只配给时长接近的分支（差 < 15%）
    if (!vSec && audioSec > 0) vSec = audioSec;   // 兜底：拿不到视频时长就用音频的
    const useAudio = !!audioWav && (vSec === 0 || Math.abs(vSec - audioSec) / Math.max(audioSec, 1) < 0.15);

    const b = { branch: name, video: p.video, duration: p.duration, raw: src, degraded, withAudio: useAudio };

    const needAudio = useAudio && !!audioWav;

    if (opt.format === 'mkv' || opt.format === 'both') {
      b.mkv = outName(opt.out, stem, name === 'h264' ? '_h264' : '_hevc', 'mkv');
      if (!mux(ffmpeg, src, needAudio ? audioWav : null, b.mkv, 'mkv', opt.fps, vSec)) { b.mkv = null; b.error = 'MKV 封装失败'; }
      else {
        const pm = probeMedia(ffmpeg, b.mkv); if (pm.duration) b.duration = pm.duration;
        if (needAudio && !audioOk(ffmpeg, b.mkv)) { b.error = '音轨时长异常(0)，请用 --audio-keep 单独取音频'; b.audioBroken = true; }
      }
    }
    if (opt.format === 'mp4' || opt.format === 'both') {
      b.mp4 = outName(opt.out, stem, name === 'h264' ? '_h264' : '_hevc', 'mp4');
      if (!toMp4Transcode(ffmpeg, src, needAudio ? audioWav : null, b.mp4, opt.fps, vSec)) { b.mp4 = null; b.error = 'MP4 转码失败'; }
      else {
        const pm = probeMedia(ffmpeg, b.mp4); if (pm.duration) b.duration = pm.duration;
        if (needAudio && !audioOk(ffmpeg, b.mp4)) { b.error = '音轨时长异常(0)'; b.audioBroken = true; }
      }
    }
    if (!opt.keep) { for (const f of new Set([raw, src])) { try { fs.unlinkSync(f); } catch (_) {} } b.raw = null; }
    return b;
  };

  if ((mode === 'h264' || mode === 'both') && d.h264) res.branches.push(buildBranch('h264', H264_TAGS, false));
  if ((mode === 'hevc' || mode === 'idr' || mode === 'both') && d.hevc) res.branches.push(buildBranch('hevc', HEVC_TAGS, true));

  // 音频单独保留（不在视频里时也有用）
  if (audioWav && !res.branches.some(b => b.withAudio)) {
    // 保留 wav
  } else if (audioWav && !opt.keep) {
    try { fs.unlinkSync(audioWav); } catch (_) {}
    if (res.audio) res.audio.wav = null;
  }

  res.ok = res.branches.length > 0 && !res.branches.every(b => b.error);
  res.seconds = +((Date.now() - t0) / 1000).toFixed(1);
  return res;
}

// ─────────────────────── 批量 ───────────────────────
function walk(dir, exts, recursive) {
  const out = [];
  const stack = [dir];
  while (stack.length) {
    const cur = stack.pop();
    let ents;
    try { ents = fs.readdirSync(cur, { withFileTypes: true }); } catch (_) { continue; }
    for (const e of ents) {
      const full = path.join(cur, e.name);
      if (e.isDirectory()) { if (recursive) stack.push(full); continue; }
      if (exts.some(x => e.name.toLowerCase().endsWith(x))) out.push(full);
    }
  }
  return out.sort();
}

function writeBatchReport(results, outDir, elapsed) {
  const okN = results.filter(r => r.ok).length;
  const lines = [
    '# 批量恢复报告', '',
    `- 扫描时间：${new Date().toLocaleString('zh-CN')}`,
    `- 输入：${opt.input}`,
    `- 输出：${outDir}`,
    `- 文件数：${results.length}（成功 ${okN}，失败 ${results.length - okN}）`,
    `- 音轨：${opt.audio ? opt.audioCodec + ' 8kHz 320采样/帧' : '关闭'}`,
    `- 总耗时：${elapsed}s`, '',
    '| # | 文件 | 大小MB | 路 | 时长 | 音轨 | 输出 | 状态 |',
    '|---|---|---|---|---|---|---|---|',
  ];
  results.forEach((r, i) => {
    if (!r.branches.length) {
      lines.push(`| ${i + 1} | ${r.stem} | ${r.sizeMB ?? '-'} | - | - | - | - | ❌ ${r.error || '无输出'} |`);
      return;
    }
    for (const b of r.branches) {
      const out = b.mp4 || b.mkv || '-';
      const au = b.withAudio ? '✅' : (r.audio ? '—' : '无');
      lines.push(`| ${i + 1} | ${r.stem} | ${r.sizeMB} | ${b.branch}${b.degraded ? '(IDR)' : ''} | ${b.duration || '-'} | ${au} | ${out} | ${b.error ? '❌ ' + b.error : '✅'} |`);
    }
  });
  const p = path.join(outDir, '_批量恢复报告.md');
  fs.writeFileSync(p, '\ufeff' + lines.join('\n'));
  fs.writeFileSync(path.join(outDir, '_批量恢复报告.json'), JSON.stringify({ input: opt.input, out: outDir, elapsed, results }, null, 2));
  return p;
}

// ─────────────────────── 主流程 ───────────────────────
(function main() {
  const t0 = Date.now();
  const isDir = fs.statSync(opt.input).isDirectory();
  opt.out = ensureOutDir(opt.out);

  if (!isDir) {
    const ffmpeg = opt.probe ? null : findFfmpeg();
    if (!opt.probe && !ffmpeg) { console.error('找不到 ffmpeg，请用 --ffmpeg 指定'); process.exit(3); }
    console.log(`\n处理 ${opt.input}`);
    const r = recoverOne(opt.input, ffmpeg);
    if (!r.ok) { console.error(`失败：${r.error || '无输出'}`); process.exit(2); }
    for (const b of r.branches) {
      console.log(`  ${b.branch}: ${b.mp4 || b.mkv || '(仅裸流)'}  ${b.video || ''} ${b.duration || ''} ${b.withAudio ? '[含音轨]' : ''}`);
    }
    if (r.audio) console.log(`  音频: ${r.audio.frames} 帧 / ${r.audio.seconds}s / ${r.audio.codec}`);
    console.log(`  耗时 ${r.seconds}s`);
    process.exit(0);
  }

  const exts = opt.ext.split(',').map(s => s.trim().toLowerCase()).filter(Boolean).map(s => s.startsWith('.') ? s : '.' + s);
  const files = walk(opt.input, exts, opt.recursive);
  if (!files.length) { console.error(`目录下没有找到 ${exts.join('/')} 文件`); process.exit(2); }

  const ffmpeg = opt.probe ? null : findFfmpeg();
  if (!opt.probe && !ffmpeg) { console.error('找不到 ffmpeg，请用 --ffmpeg 指定'); process.exit(3); }

  console.log(`\n扫描到 ${files.length} 个文件，输出到 ${opt.out}\n`);
  const results = [];
  let fail = 0;
  for (let i = 0; i < files.length; i++) {
    const f = files[i];
    process.stdout.write(`[${i + 1}/${files.length}] ${path.basename(f)} ... `);
    try {
      const r = recoverOne(f, ffmpeg);
      results.push(r);
      if (!r.ok) { fail++; console.log(`❌ ${r.error || '无输出'}`); }
      else {
        const s = r.branches.map(b => `${b.branch}:${b.duration || '?'}${b.withAudio ? '+音' : ''}`).join(' ');
        console.log(`✅ ${s}  ${r.seconds}s`);
      }
    } catch (e) {
      fail++;
      results.push({ file: f, stem: path.basename(f), ok: false, error: e.message, branches: [] });
      console.log(`❌ ${e.message}`);
    }
  }

  const elapsed = +((Date.now() - t0) / 1000).toFixed(1);
  const rp = writeBatchReport(results, opt.out, elapsed);
  console.log(`\n完成：成功 ${results.length - fail} / 失败 ${fail}，总耗时 ${elapsed}s`);
  console.log(`报告：${rp}`);
  process.exit(fail ? 4 : 0);
})();
